function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Journal"); if (!sheet) return ContentService.createTextOutput(JSON.stringify({ status: "error", trades: [] })).setMimeType(ContentService.MimeType.JSON); var rows = sheet.getDataRange().getValues(); if (rows.length <= 1) return ContentService.createTextOutput(JSON.stringify({ status: "success", trades: [] })).setMimeType(ContentService.MimeType.JSON); var headers = rows[0]; var trades = []; for (var i = 1; i < rows.length; i++) { var obj = {}; for (var j = 0; j < headers.length; j++) obj[String(headers[j]).trim()] = rows[i][j]; trades.push(obj); } return ContentService.createTextOutput(JSON.stringify({ status: "success", trades: trades })).setMimeType(ContentService.MimeType.JSON); }
